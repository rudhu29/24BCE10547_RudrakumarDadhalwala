:: =========================
"C:\Program Files\MongoDB\Server\8.3\bin\mongod.exe" --configsvr --replSet centralConfigSet --dbpath C:\data\configDB --port 27119

mongosh --port 27119

:: =========================
:: ADD SHARDS
:: =========================
sh.addShard("regionASet/localhost:27111")
sh.addShard("regionBSet/localhost:27112")

:: =========================
:: ENABLE SHARDING
:: =========================
sh.enableSharding("RetailDB")

sh.shardCollection(
  "RetailDB.sales",
  { product_id: "hashed" }
)

:: =========================
:: INSERT DATA
:: =========================
use RetailDB

for(let i=0;i<1000;i++){
    db.sales.insertOne({
        product_id: "prod_" + i,
        price: Math.random() * 500
    })
}

:: =========================
